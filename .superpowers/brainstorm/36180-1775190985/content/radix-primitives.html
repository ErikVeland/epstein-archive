Styling Pattern for Radix Primitives

Each Radix component wraps the primitive, applies lq-* BEM classes, and accepts no Tailwind. Animations use CSS keyframes, not Tailwind animate-*.

import * as RadixDialog from '@radix-ui/react-dialog';
import { cn } from '../../lib';
import './Dialog.css';
export const DialogContent = ({ className, ...props }) => (
<RadixDialog.Content
className={cn('lq-dialog__content', className)}
{...props} />
);
/* Dialog.css uses var(--glass-bg), var(--radius-xl), etc. — zero Tailwind */
